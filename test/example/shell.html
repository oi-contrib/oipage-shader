#!/bin/bash

NAME="oipage"
export PATH="$PATH:/usr/local/bin"

greet () {
    local who="$1"
    echo "hello, $who"
}

if [ "$NAME" != "" ]; then
    echo "name is $NAME"
elif [[ -n "${NAME}_x" ]]; then
    echo "fallback"
else
    echo "empty"
fi

for f in *.txt; do
    echo "$f"
done

count=0
while [ "$count" -lt 3 ]; do
    count=$((count + 1))
done

case "$NAME" in
    oipage) echo "matched" ;;
    *) echo "other" ;;
esac

result=$(greet "world")
echo "$result" | tr 'a-z' 'A-Z' > out.txt 2>&1

ls -la /tmp
gcc -o app main.c && ./app
