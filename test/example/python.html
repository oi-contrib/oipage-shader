class Point:
    def __init__(self, x, y):
        self.x = x
        self.y = y

    def __repr__(self):
        return f"Point({self.x}, {self.y})"


try:
    values: List[int] = [i * 2 for i in range(10) if i % 2 == 0]
    data = {"a": 1, "b": 2}
    total = sum(values)
    name = "world"
    greeting = f"hello, {name}!"
    square = lambda n: n ** 2
    sub = values[1:3]
    if total > 0 and name is not None:
        print(greeting, square(total), sub, data)
except ValueError as e:

finally:
