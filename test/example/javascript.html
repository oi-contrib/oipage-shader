更多情况

class Animal {
    constructor (name) {
        this.name = name;
    }
    get title () {
        return "[" + this.name + "]";
    }
    static create (name) {
        return new Animal(name);
    }
}

class Dog extends Animal {
    async run (speed = 10) {
        const parts = ["a", "b", "c"];
        const [first, ...rest] = parts;
        const msg = `dog ${this.name} runs at ${speed}`;
        const s = "he said \"hi\" and \\ backslash";
        const re = /ab+c/gi;
        const val = this?.info?.age ?? 0;
        await Promise.resolve();
        return { first, rest, msg, s, re, val };
    }
    *gen () {
        yield 1;
        yield* [2, 3];
    }
}
