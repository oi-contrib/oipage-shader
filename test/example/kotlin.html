package com.example

data class User(val name: String, val age: Int?)

object Ids {
    const val ROOT = "root"
}

fun List<Int>.secondOrNull(): Int? = this.getOrNull(1)

fun greet(user: User?): String {
    val name = user?.name ?: "guest"
    val label = when {
        user == null -> "unknown"
        user.age != null && user.age >= 18 -> "adult"
        else -> "minor"
    }

    val nums = listOf(1, 2, 3)
    val second = nums.secondOrNull()

    val raw = """
        multi-line
        text
    """.trimIndent()

    val msg = "hello, ${name} ($label), second=$second"
    return msg + raw
}

fun main() {
    println(greet(User("tom", 20)))
}
