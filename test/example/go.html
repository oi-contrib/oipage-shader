package main

import (
    "fmt"
    "strings"
)

type Greeter interface {
    Greet(name string) string
}

type Person struct {
    Name string
    Age  int
}

func (p *Person) Greet(name string) string {
    return "hi, " + name
}

func main() {
    nums := []int{1, 2, 3}
    m := map[string]int{"a": 1}
    ch := make(chan int)

    go func() {
        ch <- len(nums)
    }()

    count := <-ch
    raw := `multi
line raw`

    if count > 0 && m["a"] == 1 {
        p := &Person{Name: "tom", Age: 20}
        fmt.Println(strings.ToUpper(p.Greet("world")), raw)
    }
}
